function(yigo_add_test name)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/src/ui)
    target_link_libraries(${name} PRIVATE yigo_core Qt5::Test)
    if(${name} STREQUAL tst_boardgeometry)
        target_sources(${name} PRIVATE ${CMAKE_SOURCE_DIR}/src/ui/BoardGeometry.cpp)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()

yigo_add_test(tst_board)
yigo_add_test(tst_rules)
yigo_add_test(tst_gametree)
yigo_add_test(tst_game)
yigo_add_test(tst_sgf)
yigo_add_test(tst_boardgeometry)
yigo_add_test(tst_mainwindowlogic)
yigo_add_test(tst_gtpclient)
yigo_add_test(tst_analysisparser)
