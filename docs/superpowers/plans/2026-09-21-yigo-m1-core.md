Board.h:
```cpp
#pragma once
#include <QtGlobal>

enum class Stone { Empty, Black, White };

class Board {
public:
    static constexpr int MaxSize = 25;

    explicit Board(int size = 19);
    int size() const { return m_size; }
    bool inBounds(int x, int y) const;
    Stone stoneAt(int x, int y) const;   // 越界时 Q_ASSERT，release 返回 Empty
    static Stone opponent(Stone s);

private:
    QVector<Stone> m_grid;   // 行主序: idx = y * size + x
    int m_size = 19;
};
```

Board.cpp:
```cpp
#include "Board.h"
#include <QVector>

Board::Board(int size) : m_size(size) {
    Q_ASSERT(size >= 2 && size <= MaxSize);
    m_grid.fill(Stone::Empty, m_size * m_size);
}

bool Board::inBounds(int x, int y) const {
    return x >= 0 && x < m_size && y >= 0 && y < m_size;
}

Stone Board::stoneAt(int x, int y) const {
    if (!inBounds(x, y)) return Stone::Empty;
    return m_grid[y * m_size + x];
}

Stone Board::opponent(Stone s) {
    switch (s) {
    case Stone::Black: return Stone::White;
    case Stone::White: return Stone::Black;
    default: return Stone::Empty;
    }
}
```

Board.h 类内新增公有接口：
```cpp
public:
    bool placeStone(int x, int y, Stone color, QVector<QPoint>* capturedOut = nullptr);
    bool isLegal(int x, int y, Stone color) const;
    bool isSuicide(int x, int y, Stone color) const;
    void clear();
    void setSuperko(bool on) { m_superko = on; }
    quint64 positionHash() const { return m_hash; }

private:
    struct Chain { QVector<QPoint> stones; int liberties; };
    Chain chainAt(int x, int y) const;            // flood-fill 同色连块数气
    int libertyCount(int x, int y) const;
    void removeChain(int x, int y, QVector<QPoint>* out);
    void recomputeHash();

    QSet<quint64> m_history;   // 所有出现过的局面哈希(含当前)
    quint64 m_prevHash = 0;
    quint64 m_hash = 0;
    bool m_superko = false;
    static quint64 s_zobrist[MaxSize * MaxSize][2];
    static QRandomGenerator* s_rng;   // 固定种子初始化一次
    static void initZobrist();
```

Board.cpp 核心逻辑：
```cpp
#include <QPoint>
#include <QSet>
#include <QVector>
#include <QRandomGenerator>

quint64 Board::s_zobrist[Board::MaxSize * Board::MaxSize][2] = {};
QRandomGenerator* Board::s_rng = nullptr;

void Board::initZobrist() {
    if (s_rng) return;
    static QRandomGenerator rng(0x1BADB002u);
    s_rng = &rng;
    for (auto& cell : s_zobrist)
        for (auto& v : cell) v = rng.generate64();
}

// placeStone 步骤:
// 1. 非法检查: !inBounds || occupied(color) || isSuicide || ko → false
// 2. 写入 color; 对四邻对方连块 liberties==0 的 removeChain 提子并记录
// 3. 更新哈希: m_prevHash = m_hash; 重算 m_hash(全盘 XOR 或重扫);
//    m_history.insert(m_hash)
// 4. isLegal 判定劫: 候选局面哈希 h = 当前哈希 XOR 着点 XOR 提子效果,
//    简单劫: h == m_prevHash 拒绝; 超级劫: m_history.contains(h) 拒绝
//    实现方式: 试落子(clone 或 落子-回滚)计算候选哈希, 用"落子后重算再回滚"实现, 拒绝时无副作用
```

> 实现注意：`isLegal` 内部用 try-place-and-rollback（拷贝轻量 grid 或落子改哈希后回滚），拒绝时必须完全恢复原状态；`chainAt` 用 BFS（QQueue）+ visited QSet。

        QCOMPARE(g->board().stoneAt(7, 7), Stone::Black);
        delete g;
    }
    void compressedPointRange() {
        // AB[aa:cc] 展开: aa ab ba bb? FF4 矩形: 列 a..c 行 a..c = 9 点
        const char* sgf = "(;GM[1]FF[4]SZ[9]AB[aa:cc];W[dd])";
        auto* g = SgfParser::parse(sgf);
        QVERIFY(g != nullptr);
        int blacks = 0;
        for (int x = 0; x < 9; ++x)
            for (int y = 0; y < 9; ++y)
                if (g->board().stoneAt(x, y) == Stone::Black) ++blacks;
        QCOMPARE(blacks, 9);
        delete g;
    }
    void passMove() {
        const char* sgf = "(;GM[1]FF[4]SZ[9];B[];W[tt])";
        auto* g = SgfParser::parse(sgf);
        QVERIFY(g != nullptr);
        auto* b1 = g->tree().root()->children[0];
        QCOMPARE(b1->pos, QPoint(-1, -1));
        auto* w1 = b1->children[0];
        QCOMPARE(w1->pos, QPoint(-1, -1));   // tt = pass (9路)
        delete g;
    }
    void encodingDetection() {
        // 无 CA 声明但内容为 UTF-8 中文: 不崩溃, 注释可读
        QByteArray raw = "(;GM[1]FF[4]SZ[9]CA[UTF-8];B[cc]C[\xe4\xb8\xad\xe6\x96\x87\xe6\xa3\x8b\xe8\xb0\xb1])";
        auto* g = SgfParser::parse(QString::fromUtf8(raw));
        QVERIFY(g != nullptr);
        QCOMPARE(g->tree().root()->children[0]->comment, QString::fromUtf8("中文棋谱"));
        delete g;
    }
    void malformedInputs() {
        struct Case { const char* in; } cases[] = {
            {"", }, {"(;GM[1]", }, {";B[cc])", },           // 空/截断/无根
            {"(;GM[1];B[zz])", },                            // 非法坐标
            {"(;GM[1]FF[4])(", },                            // 多余开括号
        };
        for (auto& c : cases) {
            QString err;
            auto* g = SgfParser::parse(QString::fromLatin1(c.in), &err);
            if (g) {   // 宽松策略下部分输入可解析出空树, 不崩溃即达标
                delete g;
            } else {
                QVERIFY2(!err.isEmpty(), "error message required on failure");
            }
        }
    }
    void serializeBranchOrder() {
        // 序列化保持分支次序(先主变后变着)
        Game g(9);
        g.play(QPoint(2, 2), Stone::Black);
        auto* mainW = g.play(QPoint(3, 3), Stone::White);
        Q_UNUSED(mainW);
        g.undo();
        g.play(QPoint(4, 4), Stone::White);
        auto out = SgfParser::serialize(g);
        auto* g2 = SgfParser::parse(out);
        QVERIFY(g2 != nullptr);
        auto* b1 = g2->tree().root()->children[0];
        QCOMPARE(b1->children.size(), 2);
        QCOMPARE(b1->children[0]->pos, QPoint(3, 3));   // 主变在前
        delete g2;
    }
};

```cpp
#include <QCoreApplication>
#include <QFile>
#include <QTextStream>
#include <cstdio>
#include "Board.h"
#include "Game.h"
#include "Rules.h"
#include "SgfParser.h"

static void printBoard(const Board& b) {
    const char* glyph[] = {".", "X", "O"};
    for (int y = 0; y < b.size(); ++y) {
        for (int x = 0; x < b.size(); ++x)
            std::printf("%s ", glyph[int(b.stoneAt(x, y))]);
        std::printf("\n");
    }
}

int main(int argc, char** argv) {
    QCoreApplication app(argc, argv);
    const auto args = app.arguments();
    if (args.size() >= 2 && args[1] == "play" && args.size() >= 4) {
        Game g(args[2].toInt());
        Stone c = Stone::Black;
        for (int i = 3; i < args.size(); ++i) {
            const QString& m = args[i];
            if (m.compare("pass", Qt::CaseInsensitive) == 0) { /* pass 留 M4 */ }
            else if (m.size() >= 2) {
                int x = m[0].toUpper().toLatin1() - 'A'; if (x > 8) --x;   // 跳 I
                int y = g.boardSize() - m.mid(1).toInt();
                auto* n = g.play(QPoint(x, y), c);
                if (!n) { std::printf("illegal: %s\n", qPrintable(m)); return 1; }
                c = Board::opponent(c);
                printBoard(g.board());
                std::printf("--- move %d ---\n", n->moveNumber);
            }
        }
        Rules r(g.rules());
        auto s = r.score(g.board());
        std::printf("score: B %.1f  W %.1f  margin %.1f\n", s.blackScore, s.whiteScore, s.blackMargin);
        return 0;
    }
    if (args.size() >= 3 && args[1] == "load") {
        QFile f(args[2]);
        if (!f.open(QIODevice::ReadOnly)) { std::printf("cannot open\n"); return 1; }
        QString err;
        auto* g = SgfParser::parse(QString::fromUtf8(f.readAll()), &err);
        if (!g) { std::printf("parse error: %s\n", qPrintable(err)); return 1; }
        std::printf("size=%d nodes=%d komi=%.1f\n", g->boardSize(),
                    g->tree().nodeCount(), g->rules().komi);
        printBoard(g->board());
        delete g;
        return 0;
    }
    std::printf("usage: yigo-cli play <size> <moves...> | load <file.sgf>\n");
    return 1;
}
```
